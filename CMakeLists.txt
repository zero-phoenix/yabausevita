cmake_minimum_required(VERSION 3.10)

# --- Toolchain setup --------------------------------------------------------
# The VitaSDK ships two CMake files under share/:
#   vita.toolchain.cmake  -- the actual CMake toolchain (sets CROSS compiler,
#                           sysroot, target triple, etc.)
#   vita.cmake             -- helper module loaded via include(); provides
#                           vita_create_self(), vita_create_vpk(), etc.
# We must use vita.toolchain.cmake as CMAKE_TOOLCHAIN_FILE so the cross-
# compiler is selected before any project() call.
# ---------------------------------------------------------------------------

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    if(DEFINED ENV{VITASDK})
        set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
    else()
        message(FATAL_ERROR "Please define VITASDK to point to your SDK path!")
    endif()
endif()

project(YabauseVita C)

# movie.c and yabause.c both use this macro (normally auto-generated by
# the original autotools build, which we don't use)
add_definitions(-DVERSION="0.9.10")

# This pulls in the helper functions (vita_create_self, vita_create_vpk, etc.)
include("${VITASDK}/share/vita.cmake" REQUIRED)

set(VITA_APP_NAME "YabauseVita")
set(VITA_TITLEID  "YABA00001")
set(VITA_VERSION  "00.20")

# -----------------------------------------------------------------------
# Portable Yabause core sources (no PSP/MIPS-specific code, no platform
# GL/SDL dependencies -- see PORTING_NOTES.md for the full reasoning).
# -----------------------------------------------------------------------
set(YABAUSE_CORE_SOURCES
    src/bios.c
    src/cdbase.c
    src/cheat.c
    src/coffelf.c
    src/cs0.c
    src/cs1.c
    src/cs2.c
    src/debug.c
    src/error.c
    src/memory.c
    src/m68kcore.c
    src/m68kd.c
    src/movie.c
    src/netlink.c
    src/peripheral.c
    src/profile.c
    src/scu.c
    src/sh2core.c
    src/sh2d.c
    src/sh2idle.c
    src/sh2int.c
    src/smpc.c
    src/scsp.c
    src/vdp1.c
    src/vdp2.c
    src/vdp2debug.c
    src/vidshared.c
    src/vidsoft.c
    src/yabause.c
)

# -----------------------------------------------------------------------
# Vita platform sources: main.c (entry point) + vita_ui.c (vita2d GUI).
# The UI uses vita2d for the splash screen and the tabbed roms/bios/config
# interface (black & blue theme), then hands off to the framebuffer-based
# software renderer for emulation.
# -----------------------------------------------------------------------
set(VITA_PLATFORM_SOURCES
    src/vita/main.c
    src/vita/vita_ui.c
)

add_executable(${VITA_APP_NAME}
    ${YABAUSE_CORE_SOURCES}
    ${VITA_PLATFORM_SOURCES}
)

target_include_directories(${VITA_APP_NAME} PRIVATE src)

# -----------------------------------------------------------------------
# Link libraries. vita2d provides the 2D drawing API used by vita_ui.c
# (rectangles, PGF font, double-buffered rendering). The Sce* stubs are
# the kernel bridges vita2d and our code call into.
# -----------------------------------------------------------------------
target_link_libraries(${VITA_APP_NAME}
    vita2d
    SceDisplay_stub
    SceCtrl_stub
    SceAudio_stub
    SceSysmodule_stub
    SceGxm_stub
    SceCommonDialog_stub
    SceAppMgr_stub
    SceIofilemgr_stub
    m
)

vita_create_self(${VITA_APP_NAME}.self ${VITA_APP_NAME})

# -----------------------------------------------------------------------
# Build the VPK and embed the LiveArea assets (icon, background, splash,
# and the livearea template XML) so the app looks correct when installed
# on a Vita. The FILE syntax is:  FILE <local_path> <vpk_path>.
# -----------------------------------------------------------------------
vita_create_vpk(${VITA_APP_NAME}.vpk ${VITA_TITLEID} ${VITA_APP_NAME}.self
    VERSION ${VITA_VERSION}
    NAME ${VITA_APP_NAME}
    FILE sce_sys/icon0.png                              sce_sys/icon0.png
    FILE sce_sys/bg0.png                                sce_sys/bg0.png
    FILE sce_sys/startup.png                            sce_sys/startup.png
    FILE sce_sys/livearea/contents/template.xml         sce_sys/livearea/contents/template.xml
)
